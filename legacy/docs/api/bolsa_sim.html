 62class AportePeriodico:
 63    """DCA simplificado: aporta ``efectivo_por_sesion`` cada ``cadencia`` dias.
 64
 65    Si hay efectivo disponible en la cartera y el mercado tiene
 66    suficientes tickers, se compra una cantidad fija del ticker
 67    principal (``ticker_principal``) hasta agotar el saldo previsto.
 68    """
 69
 70    def __init__(
 71        self,
 72        ticker_principal: str,
 73        efectivo_por_sesion: float = 100.0,
 74        cadencia: int = 5,
 75    ) -> None:
 76        if efectivo_por_sesion <= 0:
 77            raise ValueError("efectivo_por_sesion debe ser positivo")
 78        if cadencia <= 0:
 79            raise ValueError("cadencia debe ser positivo")
 80        self._ticker = ticker_principal.upper()
 81        self._aporte = float(efectivo_por_sesion)
 82        self._cadencia = int(cadencia)
 83
 84    def generar_orden(
 85        self, cartera: Cartera, mercado: MercadoSimulado, sesion: int
 86    ) -> tuple[Orden, ...]:
 87        if sesion <= 0 or sesion % self._cadencia != 0:
 88            return ()
 89        if self._ticker not in mercado:
 90            return ()
 91        precio = mercado.precio_de(self._ticker)
 92        if precio <= 0 or cartera.efectivo < self._aporte:
 93            return ()
 94        cantidad = int(self._aporte // precio)
 95        if cantidad <= 0:
 96            return ()
 97        # Limitamos a lo que el efectivo permite (incluyendo comision tipica).
 98        max_por_efectivo = int((cartera.efectivo - cartera.comision_por_operacion) // precio)
 99        if max_por_efectivo <= 0:
100            return ()
101        cantidad = min(cantidad, max_por_efectivo)
102        return (Orden(TipoOperacion.COMPRA, self._ticker, cantidad),)

 89        """Valor liquidativo: efectivo + posiciones valoradas a mercado."""

 94    # -- operaciones ---------------------------------------------------------
 95    def comprar(
 96        self,
 97        mercado: MercadoSimulado,
 98        ticker: str,
 99        cantidad: int,
100        *,
101        comision: float | None = None,
102    ) -> bool:
103        """Compra ``cantidad`` participaciones. Devuelve ``True`` si se ejecuta."""
104        if isinstance(cantidad, bool) or not isinstance(cantidad, int):
105            raise TypeError("cantidad debe ser un entero")
106        if cantidad <= 0:
107            raise ValueError("cantidad debe ser positiva")
108        activo = mercado.obtener(ticker)  # KeyError si no existe
109        precio = activo.precio
110        coste = cantidad * precio
111        comision = self._comision if comision is None else float(comision)
112        coste_total = coste + comision
113        if coste_total > self._efectivo:
114            return False
115
116        posicion = self._posiciones.get(ticker)
117        if posicion is None:
118            self._posiciones[ticker] = Posicion(activo, cantidad)
119        else:
120            posicion.cantidad = posicion.cantidad + cantidad
121        self._efectivo -= coste_total
122
123        self._transacciones.append(
124            Transaccion(
125                sesion=mercado.sesion_actual,
126                tipo=TipoOperacion.COMPRA,
127                ticker=ticker,
128                cantidad=cantidad,
129                precio_unitario=precio,
130                comision=comision,
131            )
132        )
133        return True

135    def vender(
136        self,
137        mercado: MercadoSimulado,
138        ticker: str,
139        cantidad: int,
140        *,
141        comision: float | None = None,
142    ) -> bool:
143        """Vende ``cantidad`` participaciones si existen. Devuelve ``True`` si se ejecuta."""
144        if isinstance(cantidad, bool) or not isinstance(cantidad, int):
145            raise TypeError("cantidad debe ser un entero")
146        if cantidad <= 0:
147            raise ValueError("cantidad debe ser positiva")
148        posicion = self._posiciones.get(ticker)
149        if posicion is None or posicion.cantidad < cantidad:
150            return False
151        activo = mercado.obtener(ticker)
152        precio = activo.precio
153        ingreso = cantidad * precio
154        comision = self._comision if comision is None else float(comision)
155        ingreso_neto = ingreso - comision
156
157        posicion.cantidad -= cantidad
158        if posicion.cantidad == 0:
159            del self._posiciones[ticker]
160        self._efectivo += ingreso_neto
161
162        self._transacciones.append(
163            Transaccion(
164                sesion=mercado.sesion_actual,
165                tipo=TipoOperacion.VENTA,
166                ticker=ticker,
167                cantidad=cantidad,
168                precio_unitario=precio,
169                comision=comision,
170            )
171        )
172        return True

 88    def avanzar_sesion(self) -> dict[str, float]:
 89        """Avanza un dia y devuelve ``{ticker: nuevo_precio}``.
 90
 91        Para cada activo se aplica un movimiento browniano geometrico
 92        simplificado: ``factor = exp(drift + sigma * N(0,1))`` y se
 93        descuenta un tiny sesgo para realismo de retornos largos. La
 94        cota inferior ``> 0`` es un invariante de ``Activo``.
 95        """
 96        resultado: dict[str, float] = {}
 97        self._sesion_actual += 1
 98        for ticker, activo in self._activos.items():
 99            cierre_anterior = activo.precio
100            sigma = activo.volatilidad
101            # 1 / 252 ~ un dia de trading sobre un ano. Mantener este
102            # factor evita desvios explosivos en simulaciones largas.
103            dt = 1.0 / 252.0
104            shock = self._rng.gauss(0.0, 1.0)
105            # sqrt(252) invierte la anualizacion para un horizon diario.
106            factor = math.exp(self._drift * dt + sigma * math.sqrt(dt) * shock)
107            nuevo = activo.aplicar_factor(factor)
108            barra = BarraDiaria(
109                sesion=self._sesion_actual,
110                apertura=cierre_anterior,
111                cierre=nuevo,
112                maximo=max(cierre_anterior, nuevo),
113                minimo=min(cierre_anterior, nuevo),
114                volumen=0,  # se omite la dinamica de volumen en PRAC1.
115            )
116            self._historial[ticker].append(barra)
117            resultado[ticker] = nuevo
118        return resultado

35    """Calculo del maximo drawdown (caida pico-a-valle).

51            pico_drawdown = min(pico_drawdown, drawdown)
